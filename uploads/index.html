  const animFile = THINKING_ANIMATION_MAP[verb] || 'butt-brain-animated.svg';

  return (

      <img
        src={`/animations/${animFile}`}
        alt={verb}
        style={{ width: size, height: size }}
        className="object-contain"
      />
      <span className="font-semibold text-gray-100 text-sm">
        {verb}<span className="text-pink-400 animate-pulse">...</span>
      </span>
    </div>
  );
};

// Vanilla JS Dynamic Thinking Status Bar
function setThinkingVerb(verb) {
  const map = window.THINKING_VERB_MAP || {};
  const animKey = (map[verb] && map[verb].anim_id) || 'butt-brain-animated';

  // Inject embedded SVG directly from window.EMBEDDED_SVGS!
  const svgMarkup = window.EMBEDDED_SVGS[animKey];
  document.getElementById('status-icon-container').innerHTML = svgMarkup;
  document.getElementById('status-verb-label').textContent = verb;
}
