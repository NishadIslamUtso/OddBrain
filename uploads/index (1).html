Use it

<!-- standalone, anywhere (SMIL animates even in <img>) -->
<img src="animations/butt-dance.svg" width="140" height="140" alt="Boogieing…">

// React
import manifest from "./manifest.json";

const pick = (word) => manifest.words[word] ?? manifest.fallback;

export function Mascot({ word = "Thinking" }) {
  const a = manifest.animations[pick(word)];
  return (
    <span className="mascot">
      <img src={a.file} alt={word} width={120} height={120} />
      <span>{word}…</span>
    </span>
  );
}
